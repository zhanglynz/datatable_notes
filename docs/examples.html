7.1 Converting multiple columns to proportions

library(data.table)

dt <- data.table(
  ID = 1:3,
  var1 = c(2, 4, 1),
  var2 = c(3, 1, 2),
  var3 = c(5, 0, 3),
  var4 = c(0, 2, 1),
  var5 = c(0, 1, 3)
)

dt[, total := rowSums(.SD), .SDcols = var1:var5]

# this works
(new_dt <- dt[, lapply(.SD, function(x) x / total),
   .SDcols = var1:var5, by = c('ID', 'total')][, c(1, 3:7 ,2), with = FALSE])

7.4 Stratified random sampling

7.7 data.table programming

Notice using get function and setNames(list(.N), output_var_name) in the following code.

library(data.table)

my_aggregate <- function(a_dt, condi_var_nm = NULL, 
                         condi = c('1', '2'), output_var_name = NULL)
{if(is.null(condi_var_nm)) {
  return(a_dt[, .(total = .N), by = partition_code][order(partition_code)]) }
  a_dt[get(condi_var_nm) %in% condi, setNames(list(.N), output_var_name), 
       by = partition_code][order(partition_code)]   
}

set.seed(123)
fk_dt <- data.table(id = 1:60,
                    maori = sample(c('0', '1', '2'), 60, replace = TRUE),
                    asian = sample(c('0', '1', '2'), 60, replace = TRUE),
                    pacific = sample(c('0', '1', '2'), 60, replace = TRUE),
                    rand_nbr = runif(60))
fk_dt[order(rand_nbr), partition_code := (.I - 1) %% 3 + 1]

total_dt <- my_aggregate(fk_dt)
maori_dt <- my_aggregate(fk_dt, condi_var_nm = "maori", output_var_name = "maori")
asian <- my_aggregate(fk_dt, condi_var_nm = "asian", 
                      condi = '1', output_var_name = "asian")
pacific <- my_aggregate(fk_dt, condi_var_nm = "pacific", output_var_name = "pacific")

j <- 
  merge(total_dt, maori_dt, by = 'partition_code', all.x = TRUE) |>
  merge(asian, by = 'partition_code', all.x = TRUE) |>
  merge(pacific, by = 'partition_code', all.x = TRUE)
