7.1 Column Subsetting: .SDcols

Basic use

library(data.table)

dt <- data.table(ID = paste0('w', 1:5),
                 x = rnorm(5),
                 y = runif(5),
                 z = rexp(5))
(sub_dt <- dt[, .SD, .SDcols = c('ID', 'x', 'z')])
       ID           x        z
   <char>       <num>    <num>
1:     w1 -1.42110176 1.444350
2:     w2  0.48213460 1.807731
3:     w3  0.10711948 1.341875
4:     w4 -0.34061382 1.624440
5:     w5 -0.06580862 1.121430
sub_dt_2 <- dt[, .(ID, x, z)]
(identical(sub_dt, sub_dt_2))
[1] TRUE

Use is.numeric or is.character

dt[, names(.SD) := lapply(.SD, function(x) x ^2), .SDcols = is.numeric][]
       ID           x            y        z
   <char>       <num>        <num>    <num>
1:     w1 2.019530205 8.143654e-05 2.086146
2:     w2 0.232453776 3.861748e-01 3.267893
3:     w3 0.011474583 5.194679e-02 1.800629
4:     w4 0.116017776 4.029656e-01 2.638805
5:     w5 0.004330775 4.667231e-01 1.257606

Use patterns()

dt[, .SD, .SDcols = patterns('x')]
             x
         <num>
1: 2.019530205
2: 0.232453776
3: 0.011474583
4: 0.116017776
5: 0.004330775
